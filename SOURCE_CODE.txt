===== Parade_Audio.lua =====
-- Parade on Master SFX; temporarily pause/resume only the Master Music bus.
ParadeAudio={}
local active=false
local loaded=false
local pending=nil
local age=0
local backgroundPaused=false
local PLAY='Play_MilitaryParade_Internationale_V311'
local STOP='Stop_MilitaryParade_Internationale_V311'
local function EnsureLoaded()
    if loaded then return end
    if type(UI.LoadSoundBankGroup)=='function' then
        local ok,err=pcall(UI.LoadSoundBankGroup,5)
        loaded=ok
        print('[ParadeV311][AUDIO] load FMV group callOK='..tostring(ok)..' detail='..tostring(err))
    else
        print('[ParadeV311][AUDIO] LoadSoundBankGroup unavailable; using InGame load')
    end
    -- Never unload a shared bank group when closing this popup.
end
local function ResumeBackground()
    if not backgroundPaused then return end
    local ok,err=pcall(UI.PlaySound,'Resume_MilitaryParade_Background_V311')
    if ok then backgroundPaused=false end
    print('[ParadeV311][AUDIO] resume background callOK='..tostring(ok)..' detail='..tostring(err))
end
function ParadeAudio.BeginPresentation()
    EnsureLoaded()
    if backgroundPaused then return end
    local ok,err=pcall(UI.PlaySound,'Pause_MilitaryParade_Background_V311')
    backgroundPaused=ok
    print('[ParadeV311][AUDIO] pause background callOK='..tostring(ok)..' detail='..tostring(err))
end
local function Post()
    local ok,err=pcall(UI.PlaySound,PLAY)
    active=ok;age=0
    if not ok then ResumeBackground() end
    print('[ParadeV311][AUDIO] play '..PLAY..' callOK='..tostring(ok)..' detail='..tostring(err))
    return ok
end
function ParadeAudio.Stop()
    pending=nil
    if not active then return end
    local ok,err=pcall(UI.PlaySound,STOP)
    print('[ParadeV311][AUDIO] stop callOK='..tostring(ok)..' detail='..tostring(err))
    active=false
end
function ParadeAudio.EndPresentation()
    ParadeAudio.Stop()
    ResumeBackground()
end
function ParadeAudio.Play()
    ParadeAudio.BeginPresentation()
    if active or pending then
        ParadeAudio.Stop()
        pending=0.12
        return true
    end
    return Post()
end
function ParadeAudio.Tick(dt)
    if pending then
        pending=pending-dt
        if pending<=0 then pending=nil;Post() end
    elseif active then
        age=age+dt
        if age>=23.1 then active=false end
    end
end
ExposedMembers.ParadeV311Audio=ParadeAudio


===== Parade_Camera.lua =====
-- UI-only camera controller. Same APIs as installed WorldInput.lua:
-- UI.LookAtPlot / UI.GetMapZoom / UI.SetMapZoom / UI.SpinMap.
-- SpinMap takes normalized drag input, NOT degrees/radians. 1.0 is an initial
-- full-width sweep; its physical angle requires calibration in the actual game.
ParadeCamera={Duration=20,Zoom=0.0,SpinSweep=1.0}
local saved=nil
local hidden={}
local target=nil
local spinFailed=false
local refocused=false
local focusBeat=-1
local selectionSettled=false
local function Call(name,...)
    if type(UI[name])~='function' then return false end
    local ok,a,b=pcall(UI[name],...)
    if not ok then print('[ParadeV38][CAMERA] '..name..' ERROR: '..tostring(a)) end
    return ok,a,b
end
local function HideControl(path)
    local control=ContextPtr:LookUpControl(path)
    if control then
        hidden[#hidden+1]={Control=control,WasHidden=control:IsHidden()}
        control:SetHide(true)
    end
end
-- Apply focus AFTER selection/zoom. Screen positioning keeps the model in the
-- visible review area above the footer, including on wide and short displays.
local function FocusTarget()
    if not target or not target.X or not target.Y then return end
    if type(UI.LookAtPlotScreenPosition)=='function' and
        Call('LookAtPlotScreenPosition',target.X,target.Y,0.5,0.42) then return end
    Call('LookAtPlot',target.X,target.Y)
end
local function SelectTarget()
    if not target or not target.Unit then return end
    Call('DeselectAllCities')
    Call('DeselectAllUnits')
    Call('SelectUnit',target.Unit)
    -- Match WorldInput's direct-click path; stop automatic selection advancing
    -- to some other ready unit after this project's production event.
    Call('RebuildSelectionList',target.Unit)
    Call('SetCycleAdvanceTimer',0)
end
function ParadeCamera.Begin(data)
    local restarting=saved~=nil
    if not restarting then
    local host=ExposedMembers.ParadeV38UI
    local resumed=nil
    if host and host.FinishCameraRestoreNow then resumed=host.FinishCameraRestoreNow()
    elseif host and host.CancelCameraRestore then host.CancelCameraRestore() end
    saved=resumed or {}
    if not resumed then
    local selectedOK,selected=Call('GetHeadSelectedUnit')
    if selectedOK and selected then saved.UnitOwner=selected:GetOwner();saved.UnitID=selected:GetID() end
    local cityOK,selectedCity=Call('GetHeadSelectedCity')
    if cityOK and selectedCity then saved.CityOwner=selectedCity:GetOwner();saved.CityID=selectedCity:GetID() end
    local ok,zoom=Call('GetMapZoom')
    if ok and type(zoom)=='number' then saved.Zoom=zoom end
    local positioned,wx,wy=Call('GetMapLookAtWorldTarget')
    if positioned and type(wx)=='number' and type(wy)=='number' then
        local converted,x,y=Call('GetPlotCoordFromWorld',wx,wy)
        if converted then saved.X=x;saved.Y=y end
    end
    end
    end
    Call('SpinMap',0,0)
    -- These containers are distinct from AdditionalUserInterfaces, where this
    -- popup lives. Save each state; do not blindly unhide previously closed UI.
    -- Leave SelectedUnit's logic context active; hide only its visible neighbours.
    -- Selecting a unit follows the same native display path as clicking its flag.
    if not restarting then
    for _,path in ipairs({'/InGame/HUD','/InGame/TopLevelHUD','/InGame/PartialScreens',
        '/InGame/WorldViewControls/BannerAndFlags','/InGame/WorldViewControls/WorldViewIconsManager',
        '/InGame/WorldViewControls/DistrictPlotIconManager','/InGame/WorldViewControls/PlotInfo',
        '/InGame/WorldViewControls/MapPinManager','/InGame/WorldViewControls/TourismBannerManager'}) do HideControl(path) end
    local host=ExposedMembers.ParadeV38UI
    if host and host.SetPresentationHidden then host.SetPresentationHidden(true) end
    end
    -- A review always focuses the current capital's center, never an adjacent
    -- unit. Historical replay may have no garrison: keep focus on the capital.
    local candidates={}
    local player=Players[data.Owner]
    local capital=player and player:GetCities():GetCapitalCity()
    local cx=capital and capital:GetX()
    local cy=capital and capital:GetY()
    if capital then
        for _,unit in player:GetUnits():Members() do
            local row=GameInfo.Units[unit:GetType()]
            if row and ((row.Combat or 0)>0 or (row.RangedCombat or 0)>0)
                and unit:GetX()==cx and unit:GetY()==cy then
                candidates[#candidates+1]={Unit=unit,ID=unit:GetID(),X=cx,Y=cy,
                    InCity=true,Name=Locale.Lookup(row.Name)}
            end
        end
    end
    table.sort(candidates,function(a,b)return a.ID<b.ID end)
    target=candidates[1] or {X=cx,Y=cy,InCity=true,
        Name=Locale.Lookup(capital and 'LOC_PARADE_FOCUS_NO_GARRISON' or 'LOC_PARADE_NEED_CAPITAL')}
    spinFailed=false;refocused=false;focusBeat=-1;selectionSettled=false
    saved.LastSpin=0
    SelectTarget()
    Call('SpinMap',0,0)
    if target.X and target.Y then
        Call('SetMapZoom',math.max(0.3,saved.Zoom or 0.45),0,0)
        FocusTarget()
    end
    print('[ParadeV314][CAMERA] START owner='..tostring(data.Owner)..' capital='..tostring(capital and capital:GetID())..' xy='..tostring(cx)..','..tostring(cy)..' unit='..tostring(target.ID)..' cityTile='..tostring(target.InCity)..' zoom='..ParadeCamera.Zoom..' sweep='..ParadeCamera.SpinSweep)
    return target.Name,target.InCity
end
local function Smooth(v)v=math.max(0,math.min(1,v));return v*v*(3-2*v)end
function ParadeCamera.Tick(elapsed)
    if not saved or spinFailed or not target or not target.X or not target.Y then return end
    -- Finish zoom/focus work before rotating so a focus command cannot rebase yaw.
    if elapsed<=3 then
        local wide=math.max(0.3,saved.Zoom or 0.45)
        Call('SetMapZoom',wide+(ParadeCamera.Zoom-wide)*Smooth(elapsed/3),0,0)
        Call('SpinMap',0,0)
        -- The selection/production UI can enqueue a later camera move. Wait a
        -- frame, select again once, then re-anchor during the approach only.
        if elapsed>=0.25 and not selectionSettled then
            selectionSettled=true;SelectTarget()
        end
        local beat=math.floor(elapsed*5)
        if beat~=focusBeat then focusBeat=beat;FocusTarget() end
        saved.LastSpin=0
        return
    end
    if not refocused then
        refocused=true
        SelectTarget()
        Call('SetMapZoom',ParadeCamera.Zoom,0,0)
        FocusTarget()
        Call('SpinMap',0,0)
        saved.SpinStart=elapsed+0.6
    end
    -- Selection/focus transitions may overwrite the requested zoom later.
    -- Read back the live value, then reassert close-up without issuing another
    -- pan/focus command that could reset the camera transition.
    local read,actual=Call('GetMapZoom')
    if read and type(actual)=='number' then
        local diagnostic=ExposedMembers.MilitaryParadeDiagnostic or {}
        ExposedMembers.MilitaryParadeDiagnostic=diagnostic
        diagnostic.ZoomActual=actual
        if actual>ParadeCamera.Zoom+0.005 then Call('SetMapZoom',ParadeCamera.Zoom,0,0) end
    end
    local start=saved.SpinStart or 3.3
    local amount=0
    if elapsed>=start and elapsed<8 then
        amount=ParadeCamera.SpinSweep*Smooth((elapsed-start)/(8-start))
    elseif elapsed>=8 and elapsed<10.5 then
        amount=ParadeCamera.SpinSweep*(1-Smooth((elapsed-8)/2.5))
    end
    saved.LastSpin=amount
    if not Call('SpinMap',amount,0) then spinFailed=true end
end
function ParadeCamera.End()
    if not saved then return end
    -- Zero rotation on the review pivot BEFORE any selection/focus restoration.
    -- The persistent UI then waits for that return before restoring the snapshot.
    Call('SpinMap',0,0)
    for i=#hidden,1,-1 do
        local item=hidden[i]
        pcall(function()item.Control:SetHide(item.WasHidden)end)
    end
    local snapshot=saved
    hidden={};saved=nil;target=nil
    local host=ExposedMembers.ParadeV38UI
    if host and host.SetPresentationHidden then
        local ok,err=pcall(host.SetPresentationHidden,false)
        if not ok then print('[ParadeV311][CAMERA] HUD restore error: '..tostring(err)) end
    end
    -- SelectUnit/LookAtPlot can enqueue native camera work. Reset last, and
    -- finish in the persistent UI after popup dequeue/input-context removal.
    Call('SpinMap',0,0)
    if host and host.ScheduleCameraRestore then host.ScheduleCameraRestore(snapshot) end
    print('[ParadeV312][CAMERA] return on review pivot; selection/focus restore queued')
end

-- cinematic mode/engine event lock. Works when raised by the UI polling bridge.
include('InputSupport')

local pending={}
local pendingRetry=0
local current=nil
local queued=false
local started=false
local soundPlayed=false
local elapsed=0
local pushed=false
local lastCount=0
local sealPlayed=false
local function Close()
    if not current and not queued and not pushed then return end
    -- Closing is never gated by a timer, a UI bridge or camera state.
    local finished=current
    local wasShown=queued or pushed
    current=nil;started=false
    local function cleanup(label,action)
        local ok,err=pcall(action)
        if not ok then print('[ParadeV313][CLOSE] '..label..': '..tostring(err)) end
    end
    cleanup('update',function()ContextPtr:SetUpdate(nil)end)
    cleanup('audio',function()ParadeAudio.EndPresentation()end)
    cleanup('camera',function()ParadeCamera.End()end)
    if pushed then
        pushed=false;cleanup('input',function()Input.PopContext()end)
    end
    if queued then
        queued=false;cleanup('dequeue',function()UIManager:DequeuePopup(ContextPtr)end)
    end
    cleanup('hide',function()ContextPtr:SetHide(true)end)
    -- Native timeline follows the review, never obscures it. Only an actual
    -- awarded completion queues a card; replay/zero-score/duplicate close do not.
    local host=ExposedMembers.ParadeV38UI
    if host and host.QueueAwardTimeline and finished and wasShown then
        cleanup('timeline',function()host.QueueAwardTimeline(finished)end)
    end
end
local function PlayCompletionSound()ParadeAudio.Play()end
local function Phase()
    local host=ExposedMembers.ParadeV38UI
    Controls.BuildStatus:SetText(host and host.DiagnosticText and host.DiagnosticText() or '3.20 B1 | 主界面扩展未加载')
    local award=elapsed>=16
    Controls.CeremonyPanel:SetHide(elapsed<11)
    Controls.QuoteText:SetHide(award);Controls.QuoteSource:SetHide(award)
    Controls.SealPanel:SetHide(not award)
    Controls.AwardText:SetHide(not award);Controls.GeneralAward:SetHide(not award)
    Controls.SkipCeremony:SetHide(award)
    Controls.EraAward:SetHide(not award)
    local scroll=award and current and (current.EraScore or 0)>0
    Controls.EraMomentPanel:SetHide(not scroll)
    local stage=elapsed<3 and 'OPEN' or elapsed<11 and 'REVIEW' or elapsed<16 and 'QUOTE' or 'AWARD'
    Controls.StageLabel:SetText(Locale.Lookup('LOC_PARADE_STAGE_'..stage))
    if elapsed>=11 then
        local alpha=math.min(1,(elapsed-(award and 16 or 11))/0.8)
        Controls.QuoteText:SetAlpha(alpha);Controls.QuoteSource:SetAlpha(alpha)
        Controls.SealPanel:SetAlpha(alpha)
        Controls.AwardText:SetAlpha(math.min(1,math.max(0,(elapsed-16.5)/1.0)))
        Controls.GeneralAward:SetAlpha(math.min(1,math.max(0,(elapsed-17.3)/1.0)))
        Controls.SealIcon:SetOffsetVal(15-(1-alpha)*14,0)
    end
    if award and not sealPlayed then
        sealPlayed=true
        if current and not current.Replay then pcall(UI.PlaySound,'ALERT_POSITIVE') end
    end
end
local function StartScene()
    if not current or started then return end
    started=true
    ParadeAudio.BeginPresentation()
    local ok,name=pcall(ParadeCamera.Begin,current)
    if ok then Controls.FocusName:SetText(name or '')
    else print('[ParadeV38][CAMERA] BEGIN ERROR: '..tostring(name)) end
    if not current.Replay and not soundPlayed then PlayCompletionSound();soundPlayed=true end
end
local function Animate(dt)
    if not current or ContextPtr:IsHidden() then return end
    StartScene()
    if not started then return end
    elapsed=elapsed+dt
    ParadeCamera.Tick(elapsed)
    Phase()
    if elapsed>=ParadeCamera.Duration then
        ContextPtr:SetUpdate(nil)
        -- Awarded reviews flow straight into the native timeline. Zero-score
        -- diagnostics and explicit record replays retain the manual exit.
        if current and not current.Replay and (current.EraScore or 0)>0 then Close() end
    end
end
local function Show(data)
    if type(data)~='table' or data.Owner~=Game.GetLocalPlayer() then return false end
    if not data.Replay and data.Count<=lastCount then return true end
    if current then
        for _,item in ipairs(pending) do
            if not data.Replay and item.Owner==data.Owner and item.Count==data.Count then return true end
        end
        pending[#pending+1]=data
        return true
    end
    current=data;started=false;soundPlayed=false;elapsed=0;sealPlayed=false
    local quote=nil
    -- Custom GameInfo numeric indexing is not assumed to equal QuoteID.
    for row in GameInfo.ParadeQuotes() do
        if row.QuoteID==data.QuoteID then quote=row;break end
    end
    Controls.CelebrationTitle:SetText(Locale.Lookup('LOC_PARADE_CEREMONY_TITLE',data.CityName or '',data.Count))
    local signature=Locale.Lookup('LOC_PARADE_REVIEW_ARMY')
    if data.GeneralReviewed then
        local name=data.GeneralName and data.GeneralName~='' and Locale.Lookup(data.GeneralName) or Locale.Lookup('LOC_UNIT_GREAT_GENERAL_NAME')
        signature=Locale.Lookup('LOC_PARADE_REVIEW_SIGNATURE',name)
    end
    local era=data.EraName and data.EraName~='' and Locale.Lookup(data.EraName) or ''
    Controls.ReviewSignature:SetText(era..(era~='' and ' · ' or '')..signature)
    Controls.SealTitle:SetText(Locale.Lookup('LOC_PARADE_RANK_'..math.max(1,math.min(5,data.Tier or 1))))
    Phase()
    Controls.QuoteText:SetText(quote and quote.QuoteText or '')
    Controls.QuoteSource:SetText(quote and ('—— '..quote.Attribution) or '')
    local remaining=math.max(0,data.Ending-Game.GetCurrentGameTurn())
    Controls.AwardText:SetText(Locale.Lookup('LOC_PARADE_CELEBRATION_REWARD',data.Score,data.Combat,data.Loyalty,data.Amenities,remaining))
    local details=Locale.Lookup('LOC_PARADE_GENERAL_AWARD',data.GeneralPoints or 0)
    if data.GeneralReviewed then
        details=details..' · '..Locale.Lookup('LOC_PARADE_GENERAL_TIER',data.BaseTier or data.Tier or 1,data.Tier or 1)
    end
    Controls.GeneralAward:SetText(details)
    local eraTitle=(data.EraScore or 0)==2 and 'LOC_PARADE_ERA_FIRST_TITLE' or 'LOC_PARADE_ERA_REPEAT_TITLE'
    local eraBody=(data.EraScore or 0)==2 and 'LOC_PARADE_ERA_FIRST_BODY' or 'LOC_PARADE_ERA_REPEAT_BODY'
    if data.GeneralReviewed then eraBody='LOC_PARADE_ERA_GENERAL_BODY' end
    Controls.EraMomentTitle:SetText(Locale.Lookup(eraTitle))
    Controls.EraMomentBody:SetText(Locale.Lookup(eraBody))
    Controls.EraMomentSignature:SetText(signature)
    Controls.EraMomentScore:SetText(Locale.Lookup('LOC_PARADE_ERA_GAIN',data.EraScore or 0))
    local eraLine=(data.EraScore or 0)>0 and Locale.Lookup('LOC_PARADE_ERA_GAIN',data.EraScore)
        or Locale.Lookup((data.EraStatus or 3)>=3 and 'LOC_PARADE_ERA_UNAVAILABLE' or 'LOC_PARADE_ERA_ALREADY')
    Controls.EraAward:SetText(eraLine)
    Controls.CompletionTurn:SetText(Locale.Lookup('LOC_PARADE_CELEBRATION_TURN',data.Turn))
    UIManager:QueuePopup(ContextPtr,PopupPriority.High)
    queued=true
    if not data.Replay then lastCount=data.Count end
    print('[ParadeV38][CELEBRATION] QUEUED count='..tostring(data.Count))
    ContextPtr:SetUpdate(Animate)
    if not ContextPtr:IsHidden() then StartScene() end
    return true
end

local function Dispatch(data)
    local ok,result=pcall(Show,data)
    if not ok then
        print('[ParadeV38][CELEBRATION] SHOW ERROR: '..tostring(result))
        local cleaned,err=pcall(Close)
        if not cleaned then print('[ParadeV38][CELEBRATION] CLEANUP ERROR: '..tostring(err)) end
        return false
    end
    return result==true
end

    print('[ParadeV38][CIVIC] ERROR: PopulateUnlockablesForCivic unavailable')

local function ColorIcon(real)
    return setmetatable({}, {__index=function(_,key)
        if key=='SetIcon' then
            return function(_,name,...)
                if name=='ICON_PROJECT_MILITARY_PARADE' then
                    name='ICON_PROJECT_MILITARY_PARADE_COLOR'
                end
                return real:SetIcon(name,...)
            end
        end
        local value=real[key]
        if type(value)=='function' then
            return function(_,...) return value(real,...) end
        end
        return value
    end})
end
function PopulateUnlockablesForCivic(playerID,civicID,itemIM,governmentIM,...)
    if not itemIM then return basePopulate(playerID,civicID,itemIM,governmentIM,...) end
    local proxy=setmetatable({GetInstance=function()
        local instance=itemIM:GetInstance()
        return setmetatable({Icon=ColorIcon(instance.Icon)}, {__index=instance})
    end}, {__index=itemIM})
    return basePopulate(playerID,civicID,proxy,governmentIM,...)
end

local function ReadRecords(playerID)
    local result={}
    local player=Players[playerID]
    local kind=FindTemplate()
    if not player or kind==nil then return result end
    for era in GameInfo.Eras() do
        local prefix='PARADE_V314_MOMENT_'..era.Index..'_'
        local function value(key)local v=player:GetProperty(prefix..key);return v end
        if value('READY')==1 then
            result[#result+1]={ID=-1000000-era.Index,Type=kind,GameEra=era.Index,
                Turn=tonumber(value('TURN')) or 0,EraScore=tonumber(value('SCORE')) or 0,
                Parade=true,City=value('CITY') or '',General=value('GENERAL')==1,
                GeneralName=value('NAME') or '',ExtraData={},HasEverBeenCommemorated=false}
        end
    end
    local completed=tonumber((player:GetProperty('PARADE_V35_COUNT'))) or 0
    for number=1,completed do
        local prefix='PARADE_V315_MOMENT_'..number..'_'
        local function value(key)local v=player:GetProperty(prefix..key);return v end
        if value('READY')==1 then
            result[#result+1]={ID=-2000000-number,Type=kind,GameEra=tonumber(value('ERA')) or 0,
                Turn=tonumber(value('TURN')) or 0,EraScore=tonumber(value('SCORE')) or 0,
                Parade=true,City=value('CITY') or '',General=value('GENERAL')==1,
                GeneralName=value('NAME') or '',ExtraData={},HasEverBeenCommemorated=false}
        end
    end
    -- Upgrade migration: v3.13 saved only the latest award snapshot. Preserve
    -- that completion, even a repeat with +0; do not invent older award dates.
    local latestEra=tonumber((player:GetProperty('PARADE_V313_WORLD_ERA')))
    local latestScore=tonumber((player:GetProperty('PARADE_V313_ERA_SCORE'))) or 0
    if player:GetProperty('PARADE_V315_ARCHIVE_STARTED')~=1 and latestEra and latestEra>=0 and
        (latestScore>0 or player:GetProperty('PARADE_V313_ERA_DONE_'..latestEra)==1) and
        player:GetProperty('PARADE_V314_MOMENT_'..latestEra..'_READY')~=1 then
        result[#result+1]={ID=-1000000-latestEra,Type=kind,GameEra=latestEra,
            Turn=tonumber((player:GetProperty('PARADE_V35_TURN'))) or 0,EraScore=latestScore,
            Parade=true,City=player:GetProperty('PARADE_V312_CITY_NAME') or '',
            General=player:GetProperty('PARADE_V38_GENERAL')==1,
            GeneralName=player:GetProperty('PARADE_V312_GENERAL_NAME') or '',ExtraData={}}
    end
    return result
end

function AddMoment(data,isNew)
    if not data.Parade then return BaseAdd(data,isNew) end
    local first=data.EraScore==2
    local title=Locale.Lookup(first and 'LOC_PARADE_ERA_FIRST_TITLE' or 'LOC_PARADE_ERA_REPEAT_TITLE')
    local body=Locale.Lookup(data.General and 'LOC_PARADE_ERA_GENERAL_BODY' or
        (first and 'LOC_PARADE_ERA_FIRST_BODY' or 'LOC_PARADE_ERA_REPEAT_BODY'))
    local signature=Locale.Lookup(data.City)
    if data.GeneralName~='' then signature=signature..' · '..Locale.Lookup(data.GeneralName) end
    local card={}
    for k,v in pairs(data) do card[k]=v end
    card.InstanceDescription=title
    rendering=card;captured=nil
    -- Use a native large-card template to preserve the original renderer's
    -- small-moment stack boundaries and era ordering. Only its UI template is
    -- borrowed; none of its event data, illustrations or score is recorded.
    local ok,err=pcall(BaseAdd,card,isNew)
    rendering=nil
    if ok and captured then
        captured.Description:SetOffsetY(65)
        captured.Root:SetToolTipString(title..'[NEWLINE]'..signature..'[NEWLINE]'..body)
    end
    captured=nil
    if not ok then error(err) end
end
function DisplayTimeline(showAnim)
    local playerID=Game.GetLocalPlayer()
    local records=ReadRecords(playerID)
    if #records==0 then return BaseDisplay(showAnim) end
    local originalGetter=Game.GetHistoryManager
    -- Synchronous, context-local adapter. The original getter is restored even
    -- if rendering throws. Native GameCore history is never modified.
    Game.GetHistoryManager=function(...)
        local manager=originalGetter(...)
        return setmetatable({GetAllMomentsData=function(_,owner,interest)
            local original=manager:GetAllMomentsData(owner,interest)
            local merged={}
            for _,entry in ipairs(original) do merged[#merged+1]=entry end
            if owner==playerID then
                for _,entry in ipairs(records) do merged[#merged+1]=entry end
            end
            table.sort(merged,function(a,b)
                if a.Turn~=b.Turn then return a.Turn<b.Turn end
                if a.GameEra~=b.GameEra then return a.GameEra<b.GameEra end
                if (a.Parade==true)~=(b.Parade==true) then return b.Parade==true end
                return (a.ID or 0)<(b.ID or 0)
            end)
            return merged
        end},{__index=function(_,key)
            local member=manager[key]
            if type(member)=='function' then return function(_,...)return member(manager,...)end end
            return member
        end})
    end
    local ok,err=pcall(BaseDisplay,showAnim)
    Game.GetHistoryManager=originalGetter
    if not ok then
        rendering=nil;captured=nil
        print('[ParadeV314][TIMELINE] ERROR: '..tostring(err))
        -- Keep stock history usable when an incompatible timeline mod is active.
        return BaseDisplay(showAnim)
    end
    print('[ParadeV314][TIMELINE] merged parade cards='..#records)
end

-- Automatic display follows stock priority/suppression settings. Synthetic IDs
-- are consumed by the UI renderer only, never sent to GameCore notifications.
local autoSeen={}
local function RequestAuto(owner,count)
    if owner~=Game.GetLocalPlayer() or type(count)~='number' or count<=0 then return true end
    local key=tostring(owner)..':'..tostring(count)
    if autoSeen[key] then return true end
    if Options.GetAppOption('Debug','SuppressInfoPopups')==1 then return true end
    autoSeen[key]=true
    autoQueue[#autoQueue+1]={Owner=owner,Count=count}
    return true
end
local function PumpAuto(dt)
    if #autoQueue==0 then return end
    local presentation=ExposedMembers.ParadeV38Presentation
    if presentation and presentation.IsActive and presentation.IsActive() then return end
    local item=autoQueue[1]
    local player=Players[item.Owner]
    local prefix='PARADE_V315_MOMENT_'..item.Count..'_'
    if not player or player:GetProperty(prefix..'READY')~=1 then return end
    local score=tonumber((player:GetProperty(prefix..'SCORE'))) or 0
    if score<=0 then table.remove(autoQueue,1);return end
    if not item.Submitted then
        automaticOpen=true
        local ok,err=true,nil
        if ContextPtr:IsHidden() and not UIManager:IsInPopupQueue(ContextPtr) then
            ok,err=pcall(DisplayTimeline,false)
        end
        automaticOpen=false
        if not ok then
            print('[MilitaryParade][Timeline] queue failed: '..tostring(err));return
        end
        item.Submitted=true;item.Wait=0
        print('[MilitaryParade][Timeline] queued count='..item.Count..' delayed-show=false')
    end
    if ContextPtr:IsHidden() then
        item.Wait=(item.Wait or 0)+(type(dt)=='number' and dt or 0)
        -- Keep the request until it is visible. The native queue controls popup
        -- priority; a false CanShowPopup response cannot strand our own queue.
        if item.Wait>=2 and not UIManager:IsInPopupQueue(ContextPtr) then
            item.Submitted=false;item.Wait=0
            print('[MilitaryParade][Timeline] lost queue registration; retry count='..item.Count)
        end
        return
    end
    -- Start the brush animation and the award cue only after the queue has
    -- actually presented the screen. Hidden waiting time cannot consume them.
    automaticOpen=true
    local painted,err=pcall(ShowNewTimelineMoment,{momentID=-2000000-item.Count,showAnim=true})
    automaticOpen=false
    if not painted then
        print('[MilitaryParade][Timeline] visible paint retry: '..tostring(err));return
    end
    UI.PlaySound('Pride_Moment')
    diagnostic.Stage='原版年表已显示 # '..item.Count
    table.remove(autoQueue,1)
    print('[MilitaryParade][Timeline] visible count='..item.Count)
end
ExposedMembers.ParadeV316Timeline={Ready=true,RequestAuto=RequestAuto,Pump=PumpAuto}
if Events.UIIdle then Events.UIIdle.Add(PumpAuto) end

local baseGetData=GetData
local baseAdvanceProject=AdvanceProject
local paradeType='PROJECT_MILITARY_PARADE'
local function IsCurrentCapital(city)
    if not city then return false end
    local player=Players[city:GetOwner()]
    local capital=player and player:GetCities():GetCapitalCity()
    return capital~=nil and city:GetID()==capital:GetID()
end
local function CanStartParade(city)
    if not IsCurrentCapital(city) then return false end
    local player=Players[city:GetOwner()]
    local civic=GameInfo.Civics['CIVIC_EARLY_EMPIRE']
    if not civic or not player:GetCulture():HasCivic(civic.Index) then return false end
    for _,unit in player:GetUnits():Members() do
        local row=GameInfo.Units[unit:GetType()]
        if row and ((row.Combat or 0)>0 or (row.RangedCombat or 0)>0)
            and unit:GetX()==city:GetX() and unit:GetY()==city:GetY() then
            return true
        end
    end
    return false
end
function GetData(...)
    local data=baseGetData(...)
    if not data or not data.ProjectItems then return data end
    local city=data.City or UI.GetHeadSelectedCity()
    local capital=IsCurrentCapital(city)
    local canStart=capital and CanStartParade(city)
    for i=#data.ProjectItems,1,-1 do
        local item=data.ProjectItems[i]
        if item.Type==paradeType then
            if not capital then
                -- HIDE, not merely disable, even with "show disabled items" on.
                table.remove(data.ProjectItems,i)
            elseif not canStart then
                item.Disabled=true
                item.ToolTip=(item.ToolTip or '')..'[NEWLINE]'..Locale.Lookup('LOC_PARADE_NEED_GARRISON')
            end
        end
    end
    return data
end
function AdvanceProject(city,entry,...)
    if entry and entry.ProjectType==paradeType and not CanStartParade(city) then
        print('[ParadeV38][PANEL] BLOCK stale parade click')
        Refresh()
        return
    end
    return baseAdvanceProject(city,entry,...)
end
-- The stock refresh function resolves GetData dynamically. Refresh the visible
-- list when garrison or current capital changes, including after city capture.
local function OnEligibilityChanged()
    if not ContextPtr:IsHidden() then Refresh() end
end
for _,name in ipairs({'UnitMoved','UnitMoveComplete','UnitAddedToMap','UnitRemovedFromMap',
    'CityUnitsChanged','CapitalCityChanged','CityTransfered','CivicCompleted'}) do
    if Events[name] then Events[name].Add(OnEligibilityChanged) end
end

    local ability = GameInfo.UnitAbilities['ABILITY_PARADE_' .. row.Tier]
    if not building or not ability then
        print('[ParadeV38][FATAL] Missing buff building/ability for tier ' .. row.Tier)
        return
    end
    tiers[#tiers+1] = {
        Tier=row.Tier, MinimumStrength=row.MinimumStrength,
        Loyalty=row.Loyalty, Combat=row.Combat, Amenities=row.Amenities,
        BuildingIndex=building.Index, AbilityType=ability.UnitAbilityType
    }
end
table.sort(tiers,function(a,b) return a.Tier<b.Tier end)
if #tiers ~= 5 then print('[ParadeV38][FATAL] Expected 5 tiers'); return end
local function Log(s) print('[ParadeV38][Gameplay] ' .. s) end
local function Turn() return Game.GetCurrentGameTurn() end
-- An unset engine property can return ZERO Lua values, not one nil value.
-- Assignment normalizes this to one nil before tonumber receives it.
local function NumberProperty(player,key,default)
    local value=player:GetProperty(key)
    return tonumber(value) or default
end
local knownPlayers={}
local ready=false
local function RememberPlayer(playerID)
    if type(playerID)~='number' or playerID<0 then return nil end
    local player=Players[playerID]
    if player then knownPlayers[playerID]=true end
    return player
end
local function IsMilitary(unit)
    local row=GameInfo.Units[unit:GetType()]
    return row and ((row.Combat or 0)>0 or (row.RangedCombat or 0)>0)
end
local function HasGarrison(player,city)
    for _,unit in player:GetUnits():Members() do
        if IsMilitary(unit) and unit:GetX()==city:GetX() and unit:GetY()==city:GetY() then
            return true
        end
    end
    return false
end
local function GeneralPointsReward(player)
    -- Query only on a legal completion, after world-ready. No load-time player access.
    local ok,eraID=pcall(function()return player:GetEra()end)
    if not ok or type(eraID)~='number' or eraID<0 then return 5 end
    local row=GameInfo.Eras and GameInfo.Eras[eraID]
    if row and GENERAL_POINTS_BY_ERA[row.EraType] then return GENERAL_POINTS_BY_ERA[row.EraType] end
    local order=row and tonumber(row.ChronologyIndex)
    if order then
        return GENERAL_POINTS_BY_ORDER[math.max(1,math.min(#GENERAL_POINTS_BY_ORDER,math.floor(order)+1))]
    end
    return 5
end
local function ReviewGeneral(player,city)
    local found=nil
    for _,unit in player:GetUnits():Members() do
        local row=GameInfo.Units[unit:GetType()]
        if row and row.UnitType==generalClass.UnitType
            and unit:GetX()==city:GetX() and unit:GetY()==city:GetY() then
            if not found or (unit.GetID and found.GetID and unit:GetID()<found:GetID()) then found=unit end
        end
    end
    return found
end
local function HasGeneral(player,city)return ReviewGeneral(player,city)~=nil end
local function ReviewGeneralName(player,city)
    local unit=city and ReviewGeneral(player,city)
    if not unit then return '' end
    local ok,name=pcall(function()
        local individual=unit:GetGreatPerson():GetIndividual()
        local row=GameInfo.GreatPersonIndividuals and GameInfo.GreatPersonIndividuals[individual]
        return row and row.Name
    end)
    return ok and type(name)=='string' and name or 'LOC_UNIT_GREAT_GENERAL_NAME'
end
local function EraName(player)
    local ok,name=pcall(function()
        local row=GameInfo.Eras[player:GetEra()]
        return row and (row.Name or row.EraType)
    end)
    return ok and type(name)=='string' and name or ''
end
-- Native era score: first +2, then +1 for each unique legal completion.
local function AwardEraScore(player,playerID)
    if type(Game.GetEras)~='function' then return 0,3,-1 end
    local ok,eras,eraID=pcall(function()
        local manager=Game.GetEras()
        return manager,manager:GetCurrentEra()
    end)
    if not ok or not eras or type(eraID)~='number' or eraID<0
        or type(eras.ChangePlayerEraScore)~='function' then return 0,3,-1 end
    eraID=math.floor(eraID)
    local key='PARADE_V313_ERA_DONE_'..eraID
    local first=NumberProperty(player,'PARADE_V313_ERA_HAS_AWARD',0)==0
    local amount=first and 2 or 1
    local awarded,err=pcall(function()eras:ChangePlayerEraScore(playerID,amount)end)
    if not awarded then
        Log('ERA SCORE ERROR: '..tostring(err));return 0,4,eraID
    end
    player:SetProperty(key,1)
    player:SetProperty('PARADE_V313_ERA_HAS_AWARD',1)
    Log('ERA SCORE player='..playerID..' worldEra='..eraID..' amount='..amount)
    return amount,first and 1 or 2,eraID
end
local function IsEligible(player,cityID)
    if not player:GetCulture():HasCivic(civic.Index) then return false,'Early Empire incomplete' end
    local capital=player:GetCities():GetCapitalCity()
    if not capital or capital:GetID()~=cityID then return false,'not current capital' end
    if not HasGarrison(player,capital) then return false,'no own combat garrison' end
    return true
end
local function SetBuilding(city,index,wanted)
    local buildings=city:GetBuildings()
    if wanted and not buildings:HasBuilding(index) then
        city:GetBuildQueue():CreateBuilding(index)
        if not buildings:HasBuilding(index) then error('CreateBuilding failed, city='..city:GetID()..' index='..index) end
    elseif not wanted and buildings:HasBuilding(index) then
        buildings:RemoveBuilding(index)
        if buildings:HasBuilding(index) then error('RemoveBuilding failed, city='..city:GetID()..' index='..index) end
    end
end
local function ActiveTier(player)
    if Turn() >= NumberProperty(player,KEY_END,-1) then return 0 end
    return NumberProperty(player,KEY_TIER,0)
end
local function MigrateTimeline(player)
    if player:GetProperty('PARADE_V315_ARCHIVE_STARTED')==1 then return end
    local era=NumberProperty(player,'PARADE_V313_WORLD_ERA',-1)
    local score=NumberProperty(player,'PARADE_V313_ERA_SCORE',0)
    if era<0 then return end
    if score<=0 and player:GetProperty('PARADE_V313_ERA_DONE_'..era)~=1 then return end
    local prefix='PARADE_V314_MOMENT_'..era..'_'
    if player:GetProperty(prefix..'READY')==1 then return end
    player:SetProperty(prefix..'TURN',NumberProperty(player,'PARADE_V35_TURN',Turn()))
    player:SetProperty(prefix..'SCORE',score)
    player:SetProperty(prefix..'CITY',player:GetProperty('PARADE_V312_CITY_NAME') or '')
    player:SetProperty(prefix..'GENERAL',NumberProperty(player,'PARADE_V38_GENERAL',0))
    player:SetProperty(prefix..'NAME',player:GetProperty('PARADE_V312_GENERAL_NAME') or '')
    player:SetProperty(prefix..'READY',1)
end
local function SyncPlayer(player)
    MigrateTimeline(player)
    local active=ActiveTier(player)
    local refreshModifiers=player:GetProperty('PARADE_V316_MODIFIERS_VERSION')~=316
    local capital=player:GetCities():GetCapitalCity()
    local capitalID=capital and capital:GetID() or -1
    local unlocked=player:GetCulture():HasCivic(civic.Index)
    for _,city in player:GetCities():Members() do
        local allowed=unlocked and city:GetID()==capitalID and HasGarrison(player,city)
        local hadPermit=city:GetBuildings():HasBuilding(permit.Index)
        SetBuilding(city,permit.Index,allowed)
        local supervised=allowed and HasGeneral(player,city)
        local hadGeneral=city:GetBuildings():HasBuilding(generalPermit.Index)
        SetBuilding(city,generalPermit.Index,supervised)
        if hadGeneral~=supervised then Log('GENERAL city='..city:GetID()..' active='..tostring(supervised)) end
        if hadPermit~=allowed then
            Log('PERMIT city='..tostring(city:GetID())..' allowed='..tostring(allowed))
        end
        -- Removing a prerequisite does not itself revalidate an existing queue.
        -- Gameplay-only API, only on a live city, protected by SyncAll's guard.
        if hadPermit and not allowed then
            city:GetBuildQueue():CheckActiveProduction()
        end
        -- Rebuild saved helper buildings once after this modifier migration,
        -- so an already-active old save attaches the new unit-specific rows.
        if refreshModifiers then
            for _,row in ipairs(tiers) do SetBuilding(city,row.BuildingIndex,false) end
        end
        -- Remove old tier BEFORE adding new tier. Repetition cannot stack them.
        for _,row in ipairs(tiers) do
            if row.Tier~=active then SetBuilding(city,row.BuildingIndex,false) end
        end
        for _,row in ipairs(tiers) do
            if row.Tier==active then SetBuilding(city,row.BuildingIndex,true) end
        end
    end
    if refreshModifiers then player:SetProperty('PARADE_V316_MODIFIERS_VERSION',316) end
    for _,unit in player:GetUnits():Members() do
        local ability=unit:GetAbility()
        local military=IsMilitary(unit)
        for _,row in ipairs(tiers) do
            local wanted=(military and active==row.Tier) and 1 or 0
            local current=ability:GetAbilityCount(row.AbilityType) or 0
            if current~=wanted then ability:ChangeAbilityCount(row.AbilityType,wanted-current) end
        end
    end
end
local syncing=false
local function SyncAll()
    if not ready or syncing then return end
    syncing=true
    -- Only handles seen in valid player-turn callbacks; never all player slots.
    for id in pairs(knownPlayers) do
        local player=Players[id]
        if player then
            local ok,err=pcall(SyncPlayer,player)
            if not ok then Log('SYNC ERROR player='..tostring(id)..': '..tostring(err)) end
        end
    end
    syncing=false
end
local function IsParade(id)
    return id==project.Index or id==project.Hash or id==project.ProjectType
end
local function ReadUIScore(playerID)
    local ui=ExposedMembers.ParadeV3UI
    if ui and ui.GetMilitaryScore then
        local ok,value=pcall(ui.GetMilitaryScore,playerID)
        if ok and type(value)=='number' and value==value and value>=0 and value<math.huge then
            return math.floor(value)
        end
        Log('UI score unavailable: '..tostring(value))
    end
    return nil
end
local function Complete(playerID,cityID,projectID,cancelled)
    if not ready or not IsParade(projectID) or cancelled==true or cancelled==1 then return end
    local player=RememberPlayer(playerID)
    if not player then return end
    local ok,reason=IsEligible(player,cityID)
    if not ok then Log('REJECTED player='..playerID..' city='..cityID..': '..reason);return end
    -- Native and UI reporting can describe the same completion. Process it once.
    local stamp=tostring(Turn())..':'..tostring(cityID)
    if player:GetProperty(KEY_LAST)==stamp then return end
    local score=ReadUIScore(playerID)
    if score==nil then
        player:SetProperty(KEY_PENDING,cityID)
        player:SetProperty(KEY_PENDING_TURN,Turn())
        Log('PENDING completion: waiting for UI bridge, player='..playerID)
        return
    end
    local selected=tiers[1]
    for _,row in ipairs(tiers) do if score>=row.MinimumStrength then selected=row end end
    local baseTier=selected.Tier
    local capital=player:GetCities():GetCapitalCity()
    local reviewed=capital and HasGeneral(player,capital) or false
    if reviewed then selected=tiers[math.min(#tiers,baseTier+1)] end
    local previousQuote=NumberProperty(player,'PARADE_V35_QUOTE',-1)
    local choices={}
    for _,id in ipairs(quoteIDs) do if id~=previousQuote then choices[#choices+1]=id end end
    if #choices==0 then choices=quoteIDs end
    local quoteID=choices[1] or 1
    if #choices>1 then quoteID=choices[Game.GetRandNum(#choices,'Parade quotation')+1] end
    local count=NumberProperty(player,'PARADE_V35_COUNT',0)+1
    local ending=Turn()+DURATION
    player:SetProperty(KEY_SCORE,score)
    player:SetProperty(KEY_TIER,selected.Tier)
    player:SetProperty(KEY_END,ending)
    player:SetProperty(KEY_LAST,stamp)
    player:SetProperty(KEY_PENDING,-1)
    SyncPlayer(player)
    local points=0
    local rewardPoints=GeneralPointsReward(player)
    local gave,err=pcall(function()player:GetGreatPeoplePoints():ChangePointsTotal(generalClass.Index,rewardPoints)end)
    if gave then points=rewardPoints else Log('GENERAL POINTS ERROR: '..tostring(err)) end
    player:SetProperty('PARADE_V38_BASE_TIER',baseTier)
    player:SetProperty('PARADE_V38_GENERAL',reviewed and 1 or 0)
    player:SetProperty('PARADE_V38_GENERAL_POINTS',points)
    Log('GENERAL REWARD baseTier='..baseTier..' finalTier='..selected.Tier..' reviewed='..tostring(reviewed)..' points='..points)
    -- Persist the complete presentation record. COUNT is committed last so UI
    -- never observes a new completion with half-written details.
    player:SetProperty('PARADE_V35_QUOTE',quoteID)
    player:SetProperty('PARADE_V35_CITY',cityID)
    player:SetProperty('PARADE_V35_TURN',Turn())
    local cityName=capital and capital:GetName() or ''
    local generalName=ReviewGeneralName(player,capital)
    local eraName=EraName(player)
    player:SetProperty('PARADE_V312_CITY_NAME',cityName)
    player:SetProperty('PARADE_V312_GENERAL_NAME',generalName)
    player:SetProperty('PARADE_V312_ERA_NAME',eraName)
    local eraScore,eraStatus,worldEra=AwardEraScore(player,playerID)
    player:SetProperty('PARADE_V313_ERA_SCORE',eraScore)
    player:SetProperty('PARADE_V313_ERA_STATUS',eraStatus)
    player:SetProperty('PARADE_V313_WORLD_ERA',worldEra)
    -- One immutable card per completed project, committed before COUNT. UI is
    -- read-only and cannot grant a second payout when cards are opened/replayed.
    if worldEra>=0 then
        local prefix='PARADE_V315_MOMENT_'..count..'_'
        player:SetProperty(prefix..'TURN',Turn())
        player:SetProperty(prefix..'ERA',worldEra)
        player:SetProperty(prefix..'SCORE',eraScore)
        player:SetProperty(prefix..'CITY',cityName)
        player:SetProperty(prefix..'GENERAL',reviewed and 1 or 0)
        player:SetProperty(prefix..'NAME',generalName)
        player:SetProperty(prefix..'READY',1)
        player:SetProperty('PARADE_V315_ARCHIVE_STARTED',1)
    end
    player:SetProperty('PARADE_V35_COUNT',count)
    Log(string.format('AWARDED player=%d score=%d tier=%d loyalty=%d combat=%d amenities=%d expiry=%d',
        playerID,score,selected.Tier,selected.Loyalty,selected.Combat,selected.Amenities,ending))
end
local function SafeComplete(...)
    local ok,err=pcall(Complete,...)
    if not ok then Log('COMPLETION ERROR: '..tostring(err)) end
end
local function OnNativeComplete(playerID,cityID,projectID,buildingIndex,x,y,cancelled)
    if IsParade(projectID) then
        Log('NATIVE COMPLETE player='..tostring(playerID)..' city='..tostring(cityID)..' ready='..tostring(ready)..' cancelled='..tostring(cancelled))
    end
    SafeComplete(playerID,cityID,projectID,cancelled)
end
local function Refresh()
    if not ready then return end
    for playerID in pairs(knownPlayers) do
        local player=Players[playerID]
        if player then
            local pending=NumberProperty(player,KEY_PENDING,-1)
            if pending>=0 then
                if player:GetProperty(KEY_PENDING_TURN)==Turn() then
                    SafeComplete(playerID,pending,project.Index,false)
                else
                    player:SetProperty(KEY_PENDING,-1)
                    Log('Pending completion expired without UI score, player='..playerID)
                end
            end
        end
    end
    SyncAll()
end
-- UI may read this status, but receives no game-state mutation callbacks.
ExposedMembers.ParadeV3Gameplay={Version=VERSION,Ready=false}
local function OnPlayerTurnActivated(playerID)
    local player=RememberPlayer(playerID)
    if not player then return end
    if not ready then
        ready=true
        ExposedMembers.ParadeV3Gameplay.Ready=true
        Log('WORLD READY: first activated player='..tostring(playerID))
    end
    Refresh()
end
Events.CityProjectCompleted.Add(OnNativeComplete)
Events.PlayerTurnActivated.Add(OnPlayerTurnActivated)
Events.TurnBegin.Add(Refresh)
for _,name in ipairs({'UnitMoved','UnitMoveComplete','UnitTeleported','UnitAddedToMap','UnitRemovedFromMap',
    'CityUnitsChanged','CapitalCityChanged','CityProductionCompleted','CityProductionChanged','CityProductionUpdated','PlayerTurnDeactivated',
    'UnitUpgraded','CityAddedToMap','CityRemovedFromMap','CityTransfered','CivicCompleted','CitySelectionChanged'}) do
    if Events[name] then Events[name].Add(Refresh) end
end

local lastDiagnostic=''
local function EligibilityStatus(player)
    if not project then return 'LOC_PARADE_DATA_MISSING' end
    local civic=GameInfo.Civics['CIVIC_EARLY_EMPIRE']
    local permit=GameInfo.Buildings['BUILDING_PARADE_ELIGIBLE']
    local capital=player:GetCities():GetCapitalCity()
    local hasCivic=civic and player:GetCulture():HasCivic(civic.Index) or false
    if not capital then return 'LOC_PARADE_NEED_CAPITAL' end
    local hasMilitary=false
    for _,unit in player:GetUnits():Members() do
        local row=GameInfo.Units[unit:GetType()]
        if row and ((row.Combat or 0)>0 or (row.RangedCombat or 0)>0)
            and unit:GetX()==capital:GetX() and unit:GetY()==capital:GetY() then
            hasMilitary=true;break
        end
    end
    local hasPermit=permit and capital:GetBuildings():HasBuilding(permit.Index) or false
    local canProduce=false
    if project and project.Hash then
        local ok,value=pcall(function() return capital:GetBuildQueue():CanProduce(project.Hash,false) end)
        canProduce=ok and value or false
    end
    local xp2=GameInfo.Projects_XP2 and GameInfo.Projects_XP2[project.ProjectType] or nil
    local diag='capital='..tostring(capital:GetID())..' civic='..tostring(hasCivic)
        ..' ownGarrison='..tostring(hasMilitary)..' permit='..tostring(hasPermit)
        ..' canProduce='..tostring(canProduce)..' xp2Required='..tostring(xp2 and xp2.RequiredBuilding)
    if diag~=lastDiagnostic then print('[ParadeV38][UI] GATE '..diag);lastDiagnostic=diag end
    if not hasCivic then return 'LOC_PARADE_NEED_CIVIC' end
    if not hasMilitary then return 'LOC_PARADE_NEED_GARRISON' end
    if not hasPermit then return 'LOC_PARADE_WAIT_PERMIT' end
    if not canProduce then return 'LOC_PARADE_ENGINE_BLOCKED' end
    local general=GameInfo.Buildings['BUILDING_PARADE_GENERAL']
    if general and capital:GetBuildings():HasBuilding(general.Index) then return 'LOC_PARADE_AVAILABLE_GENERAL' end
    return 'LOC_PARADE_AVAILABLE'
end
local function UpdateStatus()
    if presentationHidden then Controls.ParadeStatus:SetHide(true);return end
    local bridge=ExposedMembers.ParadeV3Gameplay
    if not bridge or not bridge.Ready then Controls.ParadeStatus:SetHide(true);return end
    CheckIcons()
    local id=Game.GetLocalPlayer()
    if type(id)~='number' or id<0 then Controls.ParadeStatus:SetHide(true);return end
    local player=Players[id]
    if not player then Controls.ParadeStatus:SetHide(true);return end
    local endValue=player:GetProperty('PARADE_V3_END')
    local tierValue=player:GetProperty('PARADE_V3_TIER')
    local ending=tonumber(endValue) or -1
    local tier=tonumber(tierValue) or 0
    local remaining=ending-Game.GetCurrentGameTurn()
    if remaining<=0 or tier<=0 then
        Controls.ParadeStatus:SetText(Locale.Lookup(EligibilityStatus(player)))
        Controls.ParadeStatus:SetHide(false);return
    end
    for row in GameInfo.ParadeTiers() do
        if row.Tier==tier then
            Controls.ParadeStatus:SetText(Locale.Lookup('LOC_PARADE_ACTIVE_STATUS',
                remaining,row.Loyalty,row.Combat,row.Amenities))
            Controls.ParadeStatus:SetHide(false)
            return
        end
    end
end
-- Cancel every queued parade when the actual city conditions fail. UI uses
-- the same native BUILD / VALUE_REMOVE_AT operation as ProductionHelper.lua.
-- Never call Gameplay mutation closures across ExposedMembers.
local cleaning=false
local function CleanInvalidQueue()
    local bridge=ExposedMembers.ParadeV3Gameplay
    if cleaning or not bridge or not bridge.Ready or not project then return end
    local id=Game.GetLocalPlayer()
    if type(id)~='number' or id<0 then return end
    local player=Players[id]
    if not player then return end
    local civic=GameInfo.Civics['CIVIC_EARLY_EMPIRE']
    local capital=player:GetCities():GetCapitalCity()
    local unlocked=civic and player:GetCulture():HasCivic(civic.Index)
    cleaning=true
    local ok,err=pcall(function()
        for _,city in player:GetCities():Members() do
            local allowed=unlocked and capital and city:GetID()==capital:GetID()
            local garrison=false
            if allowed then
                for _,unit in player:GetUnits():Members() do
                    local row=GameInfo.Units[unit:GetType()]
                    if row and ((row.Combat or 0)>0 or (row.RangedCombat or 0)>0)
                        and unit:GetX()==city:GetX() and unit:GetY()==city:GetY() then
                        garrison=true;break
                    end
                end
            end
            if not allowed or not garrison then
                local queue=city:GetBuildQueue()
                -- Reverse order preserves indices; other production is untouched.
                for i=queue:GetSize()-1,0,-1 do
                    local entry=queue:GetAt(i)
                    local row=entry and entry.ProjectType~=nil and GameInfo.Projects[entry.ProjectType]
                    if row and row.ProjectType==project.ProjectType then
                        local params={}
                        params[CityOperationTypes.PARAM_INSERT_MODE]=CityOperationTypes.VALUE_REMOVE_AT
                        params[CityOperationTypes.PARAM_QUEUE_LOCATION]=i
                        CityManager.RequestOperation(city,CityOperationTypes.BUILD,params)
                        print('[ParadeV38][UI] REMOVE invalid parade city='..city:GetID()..' slot='..i)
                    end
                end
            end
        end
    end)
    cleaning=false
    if not ok then print('[ParadeV38][UI] QUEUE ERROR: '..tostring(err)) end
end
local function Refresh()
    CleanInvalidQueue()
    UpdateStatus()
end

-- Cleanup runs in this persistent UI context, including when the popup is hidden.
local cameraRestore=nil
local function ScheduleCameraRestore(snapshot)
    cameraRestore={Age=0,Step=1,Wait=0.3,Snapshot=snapshot}
end
local function CameraCleanupTick(dt)
    if not cameraRestore then return end
    local task=cameraRestore;task.Age=task.Age+dt
    local s=task.Snapshot
    -- Phase 1: stay on the review pivot and allow yaw to settle.
    if task.Step==1 and task.Age<task.Wait then pcall(UI.SpinMap,0,0);return end
    if task.Step==1 then
        pcall(function()
            UI.SpinMap(0,0)
            UI.DeselectAllUnits();UI.DeselectAllCities()
            if s.UnitOwner and s.UnitID then
                local player=Players[s.UnitOwner]
                local unit=player and player:GetUnits():FindID(s.UnitID)
                if unit then UI.SelectUnit(unit) end
            elseif s.CityOwner and s.CityID then
                local player=Players[s.CityOwner]
                local city=player and player:GetCities():FindID(s.CityID)
                if city then UI.SelectCity(city) end
            end
            if s.X and s.Y then UI.LookAtPlot(s.X,s.Y) end
            if s.Zoom then UI.SetMapZoom(s.Zoom,0,0) end
        end)
        task.Step=2;task.Wait=task.Age+0.3
        return
    end
    if task.Age>=task.Wait and task.Step==2 then
        if s.X and s.Y then pcall(UI.LookAtPlot,s.X,s.Y) end
        if s.Zoom then pcall(UI.SetMapZoom,s.Zoom,0,0) end
        pcall(UI.SpinMap,0,0);task.Step=3;task.Wait=task.Age+0.3
        return
    end
    if task.Step==3 and task.Age>=task.Wait then
        pcall(UI.SpinMap,0,0);cameraRestore=nil
        print('[ParadeV312][CAMERA] restore complete: yaw settled before selection/focus')
    end
end
-- A new completed parade must not wait on an old asynchronous restore flag.
local function FinishCameraRestoreNow()
    if not cameraRestore then return nil end
    local snapshot=cameraRestore.Snapshot
    cameraRestore=nil
    pcall(UI.SpinMap,0,0)
    if snapshot.X and snapshot.Y then pcall(UI.LookAtPlot,snapshot.X,snapshot.Y) end
    if snapshot.Zoom then pcall(UI.SetMapZoom,snapshot.Zoom,0,0) end
    return snapshot
end

local pendingAwardTimelines={}
local queuedAwardTimelines={}
local function QueueAwardTimeline(data)
    if type(data)~='table' or data.Replay or (data.EraScore or 0)<=0 then return end
    local key=tostring(data.Owner)..':'..tostring(data.Count)
    if queuedAwardTimelines[key] then return end
    queuedAwardTimelines[key]=true
    pendingAwardTimelines[#pendingAwardTimelines+1]={Owner=data.Owner,Count=data.Count}
end
local function PumpAwardTimelines(dt)
    local presentation=ExposedMembers.ParadeV38Presentation
    if presentation and presentation.IsActive and presentation.IsActive() then return end
    local timeline=ExposedMembers.ParadeV316Timeline
    if #pendingAwardTimelines>0 then
        local data=pendingAwardTimelines[1]
        local suppressed=Options and Options.GetAppOption and Options.GetAppOption('Debug','SuppressInfoPopups')==1
        local accepted=false
        if timeline and timeline.Ready then
            local ok,result=pcall(timeline.RequestAuto,data.Owner,data.Count)
            accepted=ok and result==true
        end
        if suppressed then
            diagnostic.Stage='信息弹窗已禁用'
            table.remove(pendingAwardTimelines,1)
        else
            -- This is the exact event fired by the stock timeline toolbar.
            -- Send it once even if the extension bridge was not loaded by Build.
            diagnostic.TimelineCount=data.Count
            local ok,err=pcall(function()LuaEvents.PrideMoments_ToggleTimeline(false)end)
            if ok then
                diagnostic.Stage='已发原版年表入口 # '..data.Count
                if not accepted then pcall(UI.PlaySound,'Pride_Moment') end
                table.remove(pendingAwardTimelines,1)
            else
                diagnostic.Stage='原版入口失败: '..tostring(err)
            end
        end
    end
    if timeline and timeline.Pump then
        local ok,err=pcall(timeline.Pump,dt)
        if not ok then diagnostic.Stage='年表处理失败: '..tostring(err) end
    end
end

local function CheckPresentation()
    local bridge=ExposedMembers.ParadeV3Gameplay
    if not bridge or not bridge.Ready then return end
    local id=Game.GetLocalPlayer()
    if type(id)~='number' or id<0 or not Players[id] then return end
    local data=ReadPresentation(Players[id],id)
    local count=data and data.Count or 0
    -- Establish a load baseline: reopening a save never replays old completions.
    if observedCount==nil then observedCount=count;print('[ParadeV38][UI] BASELINE count='..count) end
    if count>observedCount then
        local presentation=ExposedMembers.ParadeV38Presentation
        if presentation and presentation.Ready and presentation.Show then
            if lastFailedCount~=count or presentationClock>=retryAfter then
                print('[ParadeV38][UI] PRESENT count='..count)
                local ok,accepted=pcall(presentation.Show,data)
                if ok and accepted==true then
                    observedCount=count
                    lastFailedCount=nil
                else
                    lastFailedCount=count
                    retryAfter=presentationClock+1.5
                    print('[ParadeV38][UI] PRESENT FAILED count='..count..' detail='..tostring(accepted)..'; automatic retry in 1.5 seconds')
                end
            end
        elseif lastWaitingCount~=count then
            lastWaitingCount=count
            print('[ParadeV38][UI] WAIT PRESENTATION count='..count..'; celebration context not ready')
        end
    end
end

    presentationTimer=0

end)

#!/usr/bin/env python3
# SPDX-License-Identifier: AGPL-3.0-only
"""bnktool: read and build Wwise soundbanks (version 113) as used by Civilization VI. No Wwise installation needed.

  bnktool.py hash NAME                  FNV-1 32-bit id Wwise uses for NAME (case-insensitive): event, bank and object ids
  bnktool.py dump BANK.bnk              chunks and HIRC object list
  bnktool.py make-sfx OUTDIR --wav FILE --event NAME --control-from OFFICIAL.bnk --control-source EVENT --chain-from OFFICIAL.bnk
                                        builds a bank (+ Banks.ini entry) in which event NAME plays FILE (16-bit PCM wav). The sounds hang under a copy of
                                        the parent mixer chain of an official event (--chain-from, --chain-event); --control-from adds a second event that
                                        plays a known-good official sound from our bank
  bnktool.py clone-bank SRC.bnk OUTDIR --bank NAME --rename OLD=NEW
                                        byte copy of an official bank under a new bank id with renamed events (bisecting aid)
  bnktool.py extract SRC.bnk OUTDIR --bank NAME --event OLD --new NEW
                                        minimal bank holding one official event with its sound, parent mixers and media

--section chooses the Banks.ini section (InGame loads at game start; FMV loads on UI.LoadSoundBankGroup(5), handy for reloading without a restart).

Rules a hand-built bank must follow (found by bisecting against official banks, see docs topic 61): the DATA payload starts at a file offset that is a
multiple of 16 (BKHD is padded), the DIDX media index is sorted by id, HIRC objects are written children first (sounds, mixers bottom to top, actions,
events), sounds sit under real actor-mixers that route to a real bus. Experimental.
"""
import argparse, os, struct, sys, wave

VERSION = 113
Z = b'\x00'


def fnv(name):
    h = 2166136261
    for c in name.lower().encode():
        h = (h * 16777619) & 0xFFFFFFFF
        h ^= c
    return h


def u32(v): return struct.pack('<I', v & 0xFFFFFFFF)


# ---------------------------------------------------------------- reading
def read_chunks(data):
    i, out = 0, []
    while i + 8 <= len(data):
        tag = data[i:i+4].decode('latin1'); size = struct.unpack_from('<I', data, i + 4)[0]
        out.append((tag, i + 8, size)); i += 8 + size
    return out


def read_hirc(data):
    objs = []
    for tag, off, size in read_chunks(data):
        if tag == 'HIRC':
            n = struct.unpack_from('<I', data, off)[0]; q = off + 4
            for _ in range(n):
                t = data[q]; s = struct.unpack_from('<I', data, q + 1)[0]
                objs.append((t, data[q + 5:q + 5 + s])); q += 5 + s
    return objs


HIRC_TYPES = {1: 'settings', 2: 'sound', 3: 'action', 4: 'event', 5: 'random/seq', 6: 'switch', 7: 'actor-mixer', 8: 'bus', 9: 'layer',
              10: 'music-segment', 11: 'music-track', 12: 'music-switch', 13: 'music-playlist', 14: 'attenuation', 18: 'state', 19: 'fx', 20: 'aux-bus'}


def cmd_dump(a):
    data = open(a.bank, 'rb').read()
    for tag, off, size in read_chunks(data):
        extra = ''
        if tag == 'BKHD':
            ver, bid = struct.unpack_from('<II', data, off); extra = 'version %d bank id %d' % (ver, bid)
        print('%-4s offset %-9d size %-9d %s' % (tag, off, size, extra))
    for t, body in read_hirc(data):
        print('  %-14s id %-11d %d bytes' % (HIRC_TYPES.get(t, 'type %d' % t), struct.unpack_from('<I', body, 0)[0], len(body)))


# ---------------------------------------------------------------- building
def pcm_wem(wav_path):
    """Wwise PCM media: RIFF/WAVE with an extensible fmt chunk, like the game's own PCM .wem files."""
    w = wave.open(wav_path, 'rb')
    ch, rate, width, frames = w.getnchannels(), w.getframerate(), w.getsampwidth(), w.getnframes()
    if width != 2: sys.exit('only 16-bit PCM wav is supported')
    pcm = w.readframes(frames)
    mask = {1: 4, 2: 3}.get(ch)
    if mask is None: sys.exit('only mono or stereo wav is supported')
    fmt = struct.pack('<HHIIHHH', 0xFFFE, ch, rate, rate * ch * 2, ch * 2, 16, 6) + struct.pack('<HI', 0, mask)
    akd = struct.pack('<ffII', 1.0, 1.0, 0, 0)
    body = b'WAVE' + b'fmt ' + u32(len(fmt)) + fmt + b'akd ' + u32(len(akd)) + akd + b'JUNK' + u32(12) + bytes(12) + b'data' + u32(len(pcm)) + pcm
    return b'RIFF' + u32(len(body)) + body


# Byte templates from Cinematic_Tutorials_MusicSFX.bnk. Ids/links are patched in by build_bank().
SOUND_TAIL = bytes.fromhex('000000c000040101000000000000000000')   # after the parent id
# exact template bodies (hex) from the game
T_MIXER = '04c6312300000000000000a87723210001000000104100c000000100000000000000000000010000002c261d01'
T_WORKUNIT = '5f4c193500000037bcb7e2000000000001000000803f00c30000010000000000000000000001000000a8772321'


def official_sound(bank_path, sound_id=None, min_media=0):
    """Pick the smallest in-bank (stream type 0) sound of an official bank: returns (body, media bytes). Used as a control."""
    data = open(bank_path, 'rb').read()
    didx, dstart = {}, 0
    for tag, off, size in read_chunks(data):
        if tag == 'DIDX':
            for i in range(size // 12):
                mid, o, sz = struct.unpack_from('<III', data, off + i * 12); didx[mid] = (o, sz)
        if tag == 'DATA': dstart = off
    best = None
    for t, body in read_hirc(data):
        if t != 2 or len(body) != 46: continue
        oid, plugin, stream, src, msz = struct.unpack_from('<IIBII', body, 0)
        if stream != 0 or src not in didx or body[18:21] != bytes(3): continue
        o, sz = didx[src]
        if sound_id is not None and oid != sound_id: continue
        if sz < min_media: continue
        if best is None or sz < best[2]: best = (body, data[dstart + o:dstart + o + sz], sz)
    if best is None: sys.exit('no suitable sound found in ' + bank_path)
    return best[0], best[1]


def official_sound_of_event(bank_path, event):
    """(sound body, media bytes) of the sound that an official event plays (the event must have one play action targeting an in-bank sound)."""
    data = open(bank_path, 'rb').read()
    objs = {struct.unpack_from('<I', body, 0)[0]: (t, body) for t, body in read_hirc(data)}
    ev = objs[fnv(event)][1]
    act = objs[struct.unpack_from('<I', ev, 8)[0]][1]
    snd = objs[struct.unpack_from('<I', act, 6)[0]][1]
    src = struct.unpack_from('<I', snd, 9)[0]
    for tag, off, size in read_chunks(data):
        if tag == 'DIDX':
            for i in range(size // 12):
                mid, o, sz = struct.unpack_from('<III', data, off + i * 12)
                if mid == src: didx = (o, sz)
        if tag == 'DATA': dstart = off
    return snd, data[dstart + didx[0]:dstart + didx[0] + didx[1]]


def official_chain(bank_path, event):
    """Bodies of the actor-mixers above the sound that an official event plays (bottom to top), to be reused as the parent chain of our own sounds."""
    data = open(bank_path, 'rb').read()
    objs = {struct.unpack_from('<I', body, 0)[0]: (t, body) for t, body in read_hirc(data)}
    ev = objs[fnv(event)][1]
    act = objs[struct.unpack_from('<I', ev, 8)[0]][1]
    snd = objs[struct.unpack_from('<I', act, 6)[0]][1]
    chain, parent = [], struct.unpack_from('<I', snd, 25)[0]
    while parent:
        body = objs[parent][1]
        chain.append(body)
        parent = mixer_parent(body)[0]
    return chain


def build_bank(name, voices, align=16, chain=None):
    """voices: list of dicts {event, wem, body (optional: official sound body to clone)}.
    Parent objects: our own simple work unit > mixer, or (chain=official_chain(...)) a copy of an official parent chain with new ids."""
    bank_id = fnv(name)
    mixer_id = fnv(name + ('/chain0' if chain else '/mixer'))
    wu_id = fnv(name + '/workunit')
    objs, media, sound_ids, ids = [], [], [], {'bank': bank_id, 'events': {}}
    sounds_l, actions_l, events_l = [], [], []   # official banks list children before their parents: sounds, mixers (bottom to top), actions, events
    for i, v in enumerate(voices):
        media_id = fnv('%s/media%d' % (name, i)); sound_id = fnv('%s/sound%d' % (name, i)); action_id = fnv('%s/play%d' % (name, i))
        wem = v['wem']
        if v.get('body'):
            sb = bytearray(v['body'])
            sb[0:4] = u32(sound_id); sb[9:13] = u32(media_id); sb[25:29] = u32(mixer_id)
            if 'plugin' in v: sb[4:8] = u32(v['plugin']); sb[13:17] = u32(len(wem))   # same sound object, other codec (e.g. PCM)
            sound = bytes(sb)
        else:
            sound = u32(sound_id) + u32(0x00010001) + Z + u32(media_id) + u32(len(wem)) + Z + bytes(3) + u32(0) + u32(mixer_id) + SOUND_TAIL
        action = u32(action_id) + b'\x03\x04' + u32(sound_id) + bytes(3) + b'\x04' + u32(bank_id)
        evt = u32(fnv(v['event'])) + u32(1) + u32(action_id)
        sounds_l.append((2, sound)); actions_l.append((3, action)); events_l.append((4, evt))
        media.append((media_id, wem)); sound_ids.append(sound_id); ids['events'][v['event']] = fnv(v['event'])
    if chain:
        cids = [fnv('%s/chain%d' % (name, i)) for i in range(len(chain))]
        parents = []
        for i, body in enumerate(chain):
            head, _ = split_children(body)
            h = bytearray(head)
            h[0:4] = u32(cids[i])
            orig_parent, pq = mixer_parent(body)
            h[pq:pq + 4] = u32(cids[i + 1] if i + 1 < len(cids) else orig_parent)
            kids = sound_ids if i == 0 else [cids[i - 1]]
            parents.append((7, bytes(h) + u32(len(kids)) + b''.join(u32(x) for x in kids)))
        objs = sounds_l + parents + actions_l + events_l
    else:
        mixer = bytearray(bytes.fromhex(T_MIXER)[:-8])
        mixer[0:4] = u32(mixer_id); mixer[11:15] = u32(wu_id); mixer[18:22] = struct.pack('<f', 0.0)
        mixer = bytes(mixer) + u32(len(sound_ids)) + b''.join(u32(x) for x in sound_ids)
        wu = bytearray(bytes.fromhex(T_WORKUNIT))
        wu[0:4] = u32(wu_id); wu[-4:] = u32(mixer_id)
        objs = sounds_l + [(7, mixer), (7, bytes(wu))] + actions_l + events_l
    return assemble(name, objs, media, align), ids


def assemble(name, objs, media, align=16):
    """objs: [(hirc type, body)], media: [(media id, bytes)] -> bank file bytes."""
    bank_id = fnv(name)
    hirc = u32(len(objs)) + b''.join(bytes([t]) + u32(len(b)) + b for t, b in objs)
    data, entries = b'', []
    for mid, wem in media:
        entries.append((mid, len(data), len(wem)))
        data += wem + bytes((-len(wem)) % align)
    # the media index is sorted by id in official banks (Wwise looks media up by binary search)
    didx = b''.join(u32(m) + u32(o) + u32(s) for m, o, s in sorted(entries))
    bkhd = u32(VERSION) + u32(bank_id) + u32(0) + u32(0) + u32(4)
    def chunk(tag, body): return tag.encode() + u32(len(body)) + body
    if not media:   # banks without media have no DIDX / DATA chunks
        return chunk('BKHD', bkhd + u32(0)) + chunk('HIRC', hirc)
    # Official banks start the DATA payload at a file offset that is a multiple of 16: Wwise pads the BKHD tail to get there.
    while (8 + len(bkhd) + 8 + len(didx) + 8) % align:
        bkhd += u32(0)
    return chunk('BKHD', bkhd) + chunk('DIDX', didx) + chunk('DATA', data) + chunk('HIRC', hirc)


def split_children(body):
    """Actor-mixer body -> (head, [child ids]); the child list is the tail: u32 count + ids."""
    for k in range(1, 400):
        pos = len(body) - 4 - 4 * k
        if pos >= 0 and struct.unpack_from('<I', body, pos)[0] == k: return body[:pos], [struct.unpack_from('<I', body, pos + 4 + 4 * i)[0] for i in range(k)]
    sys.exit('cannot find the child list')


def mixer_parent(body):
    """DirectParentID of an actor-mixer: id, ovFx, nfx, [bitsBypass + 7 bytes per fx], ovAttach, bus, parent."""
    q = 4; q += 1; nfx = body[q]; q += 1
    if nfx: q += 1 + 7 * nfx
    q += 1 + 4
    return struct.unpack_from('<I', body, q)[0], q


def cmd_extract(a):
    """Minimal bank holding one official event: event, play action, sound, and the sound's real parent mixers (child lists trimmed), plus its media."""
    data = open(a.src, 'rb').read()
    objs = {struct.unpack_from('<I', body, 0)[0]: (t, body) for t, body in read_hirc(data)}
    didx, dstart = {}, 0
    for tag, off, size in read_chunks(data):
        if tag == 'DIDX':
            for i in range(size // 12):
                mid, o, sz = struct.unpack_from('<III', data, off + i * 12); didx[mid] = (o, sz)
        if tag == 'DATA': dstart = off
    ev = objs[fnv(a.event)][1]
    act_id = struct.unpack_from('<I', ev, 8)[0]
    act = objs[act_id][1]
    sound_id = struct.unpack_from('<I', act, 6)[0]
    snd = objs[sound_id][1]
    src = struct.unpack_from('<I', snd, 9)[0]
    out = [(4, struct.pack('<II', fnv(a.new), 1) + u32(act_id)), (3, act), (2, snd)]
    child, parent = sound_id, struct.unpack_from('<I', snd, 25)[0]
    while parent:
        t, body = objs[parent]
        head, kids = split_children(body)
        body = head + u32(1) + u32(child)
        out.append((t, body))
        child = parent
        parent = mixer_parent(body)[0]
    o, sz = didx[src]
    blob = assemble(a.bank, out, [(src, data[dstart + o:dstart + o + sz])])
    os.makedirs(a.outdir, exist_ok=True)
    open(os.path.join(a.outdir, a.bank + '.bnk'), 'wb').write(blob)
    ini_path = os.path.join(a.outdir, a.ini)
    lines = open(ini_path, encoding='utf-8').read().splitlines() if os.path.exists(ini_path) else ['[Global]', '', '[Menu]', '', '[InGame]', '', '[2D]', '', '[3D]', '', '[FMV]']
    if (a.bank + '.bnk') not in lines: lines.insert(lines.index('[%s]' % a.section) + 1, a.bank + '.bnk')
    open(ini_path, 'w', newline='\r\n').write('\n'.join(lines) + '\n')
    print('wrote %s.bnk (%d bytes): %d objects, media %d (%d bytes), event %s' % (a.bank, len(blob), len(out), src, sz, a.new))

def cmd_clone_bank(a):
    """Copy an official bank under a new bank id and rename events (old name -> new name). A bisecting aid: a byte-exact clone with only ids changed."""
    data = bytearray(open(a.src, 'rb').read())
    chunks = read_chunks(bytes(data))
    for tag, off, size in chunks:
        if tag == 'BKHD': data[off + 4:off + 8] = u32(fnv(a.bank))
    renames = {fnv(old): fnv(new) for old, new in (r.split('=') for r in a.rename)}
    done = 0
    for tag, off, size in chunks:
        if tag != 'HIRC': continue
        n = struct.unpack_from('<I', data, off)[0]; q = off + 4
        for _ in range(n):
            t = data[q]; s = struct.unpack_from('<I', data, q + 1)[0]
            if t == 4:
                oid = struct.unpack_from('<I', data, q + 5)[0]
                if oid in renames: data[q + 5:q + 9] = u32(renames[oid]); done += 1
            q += 5 + s
    if done != len(renames): sys.exit('only %d of %d events found' % (done, len(renames)))
    os.makedirs(a.outdir, exist_ok=True)
    open(os.path.join(a.outdir, a.bank + '.bnk'), 'wb').write(bytes(data))
    ini_path = os.path.join(a.outdir, a.ini)
    lines = open(ini_path, encoding='utf-8').read().splitlines() if os.path.exists(ini_path) else ['[Global]', '', '[Menu]', '', '[InGame]', '', '[2D]', '', '[3D]', '', '[FMV]']
    if (a.bank + '.bnk') not in lines:
        lines.insert(lines.index('[%s]' % a.section) + 1, a.bank + '.bnk')
    open(ini_path, 'w', newline='\r\n').write('\n'.join(lines) + '\n')
    print('wrote %s.bnk (%d bytes) bank id %d, renamed %d event(s)' % (a.bank, len(data), fnv(a.bank), done))


def parse_switch_tree(body, groups=2):
    """Music switch container body -> (head bytes before the tree size field, mode, [nodes]) for a decision tree of depth 2.
    Layout (v113): ... u32 depth, u32 group ids[depth], u8 group types[depth], u32 tree size, u8 mode, nodes (12 bytes each: key u32, audio id or
    (u16 idx, u16 count), u16 weight, u16 probability). The tree is the tail of the object. Nodes are breadth first, siblings sorted by key."""
    depth = None
    for pos in range(0, len(body) - 8):
        if struct.unpack_from('<I', body, pos)[0] == groups and pos + 4 + 4 * groups + groups + 5 < len(body):
            q = pos + 4 + 4 * groups + groups
            size = struct.unpack_from('<I', body, q)[0]
            if size % 12 == 0 and q + 5 + size == len(body) and size >= 12 * (groups + 1): depth = pos; break
    if depth is None: sys.exit('cannot find the decision tree (only depth-2 music switch containers are supported)')
    q = depth + 4 + 4 * groups + groups
    size = struct.unpack_from('<I', body, q)[0]
    nodes = [list(struct.unpack_from('<IIHH', body, q + 5 + 12 * i)) for i in range(size // 12)]
    return body[:q], body[q + 4], nodes, [struct.unpack_from('<I', body, depth + 4 + 4 * i)[0] for i in range(groups)]


def cmd_add_civ(a):
    """Add a civilization (switch value) to a music switch container: a copy of the container with one more leaf under every era, mapped to the
    same music as an existing civilization (--like). Writes a bank that holds only the modified container."""
    data = open(a.src, 'rb').read()
    objs = {struct.unpack_from('<I', body, 0)[0]: (t, body) for t, body in read_hirc(data)}
    t, body = objs[a.container]
    if t != 12: sys.exit('object %d is not a music switch container' % a.container)
    head, mode, nodes, groups = parse_switch_tree(body)
    root = nodes[0]
    r_idx, r_cnt = root[1] & 0xFFFF, root[1] >> 16
    eras = nodes[r_idx:r_idx + r_cnt]
    new_key, like_key = (fnv(a.civ), fnv(a.like)) if a.civ else (None, None)
    swap = [(fnv(x), fnv(y)) for x, y in (r.split('=') for r in a.remap)]
    blocks = []
    for era in eras:
        i, c = era[1] & 0xFFFF, era[1] >> 16
        kids = [list(k) for k in nodes[i:i + c]]
        for dst, src in swap:   # existing civilization takes the music of another one (an override test)
            d = next(k for k in kids if k[0] == dst); sk = next(k for k in kids if k[0] == src); d[1] = sk[1]
        if a.civ:
            if any(k[0] == new_key for k in kids): sys.exit('%s already exists' % a.civ)
            like = next((k for k in kids if k[0] == like_key), None)
            if like is None: sys.exit('%s not found under every era' % a.like)
            kids.append([new_key, like[1], like[2], like[3]])
            kids.sort(key=lambda k: k[0])
        blocks.append(kids)
    out = [list(root)]
    out[0][1] = 1 | (len(eras) << 16)
    nxt = 1 + len(eras)
    era_nodes = []
    for era, kids in zip(eras, blocks):
        era_nodes.append([era[0], nxt | (len(kids) << 16), era[2], era[3]])
        nxt += len(kids)
    out += era_nodes
    for kids in blocks: out += kids
    tree = b''.join(struct.pack('<IIHH', *n) for n in out)
    new_body = head + u32(len(tree)) + bytes([mode]) + tree
    blob = assemble(a.bank, [(12, new_body)], [])
    os.makedirs(a.outdir, exist_ok=True)
    open(os.path.join(a.outdir, a.bank + '.bnk'), 'wb').write(blob)
    ini_path = os.path.join(a.outdir, a.ini)
    lines = open(ini_path, encoding='utf-8').read().splitlines() if os.path.exists(ini_path) else ['[Global]', '', '[Menu]', '', '[InGame]', '', '[2D]', '', '[3D]', '', '[FMV]']
    if (a.bank + '.bnk') not in lines: lines.insert(lines.index('[%s]' % a.section) + 1, a.bank + '.bnk')
    open(ini_path, 'w', newline='\r\n').write('\n'.join(lines) + '\n')
    print('wrote %s.bnk (%d bytes): container %d now has %d nodes (was %d); civ "%s" (key %s) plays like "%s" under %d eras; remaps %s' % (a.bank, len(blob), a.container, len(out), len(nodes), a.civ, new_key, a.like, len(eras), a.remap))


# Switch container (HIRC type 6) template: Unit_Bank_3D.bnk object 30990b2f (Game_Location, four era switch values, one child).
# Layout (v113): id, NodeBaseParams (parent at offset 11), u8 group type, u32 group id, u32 default switch, u8 continuous validation,
# u32 child count + ids, u32 switch count (per switch: u32 switch id, u32 item count + node ids), u32 param count (per node: u32 id, u8, u8, i32 fade out, i32 fade in).
T_SWITCH = ('30990b2f000000000000006ed80f1f0001000000c0c200c0000001000000000000000000000097df40167c02792a00010000006e6d2b32040000007c02792a010000006e6d2b32'
            'db89329e010000006e6d2b32d7f21f82010000006e6d2b328f70a83a010000006e6d2b32020000006e6d2b32000100000000000000009b6f903500010000000000000000')


def switch_container(container_id, parent_id, group, values, children):
    """values: [(switch value name, [child node ids])]; children: all child ids. Built from T_SWITCH by replacing its group, switch and child lists."""
    t = bytes.fromhex(T_SWITCH)
    gpos = t.find(u32(fnv('Game_Location')))
    head = bytearray(t[:gpos - 1])   # id .. NodeBaseParams (the byte before the group id is the group type)
    head[0:4] = u32(container_id); head[11:15] = u32(parent_id)
    assert head[16] == 1 and head[17] == 0   # one property, the volume
    head[18:22] = struct.pack('<f', 0.0)     # the template's own volume is -96 dB (silent)
    gtype = t[gpos - 1:gpos]
    default = u32(fnv(values[0][0]))
    out = bytes(head) + gtype + u32(fnv(group)) + default + Z + u32(len(children)) + b''.join(u32(c) for c in children)
    out += u32(len(values))
    for name, nodes in values: out += u32(fnv(name)) + u32(len(nodes)) + b''.join(u32(n) for n in nodes)
    out += u32(len(children)) + b''.join(u32(c) + b'\x00\x01' + u32(0) + u32(0) for c in children)
    return out


def cmd_make_switch(a):
    """A bank with one switch container (new ids) that plays a different sound for each value of a global switch group, e.g. Game_Location (era)."""
    sound_src, chain_src = a.sounds_from, a.chain_from
    body, _ = official_sound_of_event(sound_src, a.sound_source)
    chain = official_chain(chain_src, a.chain_event)
    name = a.bank
    bank_id = fnv(name)
    cont_id = fnv(name + '/switch')
    cids = [fnv('%s/chain%d' % (name, i)) for i in range(len(chain))]
    values, objs_s, media, children = [], [], [], []
    for i, spec in enumerate(a.value):
        vname, wav = spec.split('=', 1)
        wem = pcm_wem(wav)
        sid, mid = fnv('%s/sound%d' % (name, i)), fnv('%s/media%d' % (name, i))
        sb = bytearray(body)
        sb[0:4] = u32(sid); sb[9:13] = u32(mid); sb[25:29] = u32(cont_id); sb[4:8] = u32(0x00010001); sb[13:17] = u32(len(wem))
        objs_s.append((2, bytes(sb))); media.append((mid, wem)); children.append(sid); values.append((vname, [sid]))
    parents = []
    for i, cb in enumerate(chain):
        head, _ = split_children(cb)
        h = bytearray(head); h[0:4] = u32(cids[i])
        orig_parent, pq = mixer_parent(cb)
        h[pq:pq + 4] = u32(cids[i + 1] if i + 1 < len(cids) else orig_parent)
        kids = [cont_id] if i == 0 else [cids[i - 1]]
        parents.append((7, bytes(h) + u32(len(kids)) + b''.join(u32(x) for x in kids)))
    cont = switch_container(cont_id, cids[0], a.group, values, children)
    action_id = fnv(name + '/play')
    action = u32(action_id) + b'\x03\x04' + u32(cont_id) + bytes(3) + b'\x04' + u32(bank_id)
    evt = u32(fnv(a.event)) + u32(1) + u32(action_id)
    objs = objs_s + [(6, cont)] + parents + [(3, action), (4, evt)]
    blob = assemble(name, objs, media)
    os.makedirs(a.outdir, exist_ok=True)
    open(os.path.join(a.outdir, name + '.bnk'), 'wb').write(blob)
    ini_path = os.path.join(a.outdir, a.ini)
    lines = open(ini_path, encoding='utf-8').read().splitlines() if os.path.exists(ini_path) else ['[Global]', '', '[Menu]', '', '[InGame]', '', '[2D]', '', '[3D]', '', '[FMV]']
    if (name + '.bnk') not in lines: lines.insert(lines.index('[%s]' % a.section) + 1, name + '.bnk')
    open(ini_path, 'w', newline='\r\n').write('\n'.join(lines) + '\n')
    print('wrote %s.bnk (%d bytes): event "%s" id %d plays a switch container on %s with %d values' % (name, len(blob), a.event, fnv(a.event), a.group, len(values)))

#!/usr/bin/env python3
# SPDX-License-Identifier: AGPL-3.0-only
"""Build only the parade's native v113 bank. No game DLL or Wwise editor required.
Routing template and bnktool: cru121/civ6-dev-ce (AGPL-3.0-only).
"""
from pathlib import Path
import json, struct
import bnktool as b

ROOT = Path(__file__).resolve().parents[1]
NAME = 'MilitaryParade_Internationale_V311_Bank'
PLAY = 'Play_MilitaryParade_Internationale_V311'
STOP = 'Stop_MilitaryParade_Internationale_V311'

def build():
    template = json.loads((ROOT/'tools/parade_audio_template.json').read_text())
    wav = ROOT/'AudioSource/Parade_Internationale.wav'
    # A single private 2D mixer routes straight to the stock Master SFX bus.
    # Discard the copied notification mixers, FX/state references and their buses.
    mixer = bytearray(bytes.fromhex(template['chain'][0]))
    assert len(mixer) == 44 and mixer[5] == 0
    mixer[4] = 1  # override parent FX, with no FX
    mixer[7:11] = b.u32(3577355056)
    mixer[11:15] = b.u32(0)
    mixer[18] = 0xC3  # explicit 2D positioning, no 3D attenuation
    blob, _ = b.build_bank(NAME, [dict(event=PLAY, wem=b.pcm_wem(str(wav)),
        body=bytes.fromhex(template['sound']), plugin=0x00010001)],
        chain=[bytes(mixer)])
    objects = b.read_hirc(blob)
    sound = next(body for kind, body in objects if kind == 2)
    target = struct.unpack_from('<I', sound)[0]
    stop_id = b.fnv(NAME+'/stop')
    # Wwise v113 Stop_E, target-specific across game objects. No Stop_ALL.
    # id, actionType, target, isBus, empty properties/ranges, linear fade, exceptions.
    stop = b.u32(stop_id) + struct.pack('<H',0x0102) + b.u32(target) + bytes(3) + b'\x04' + b.u32(0)
    # A play event only plays. Replay stops in Lua and waits a frame before play.
    events = list(objects)
    idx = next(i for i,(kind,_) in enumerate(events) if kind == 4)
    events.insert(idx,(3,stop))
    events.append((4,b.u32(b.fnv(STOP))+b.u32(1)+b.u32(stop_id)))
    # Pause/resume only the stock Master Music bus. Parade audio uses Master SFX.
    # Keep the normal resume count semantics (bIsMasterResume=0).
    for label,atype in [('Pause',0x0202),('Resume',0x0302)]:
        aid=b.fnv(NAME+'/'+label.lower()+'_background')
        action=b.u32(aid)+struct.pack('<H',atype)+b.u32(48433064)+b'\x01'+bytes(2)+b'\x04'+bytes(1)+b.u32(0)
        idx=next(i for i,(kind,_) in enumerate(events) if kind==4)
        events.insert(idx,(3,action))
        ename=label+'_MilitaryParade_Background_V311'
        events.append((4,b.u32(b.fnv(ename))+b.u32(1)+b.u32(aid)))
    chunks = {tag:(off,size) for tag,off,size in b.read_chunks(blob)}
    off,size = chunks['DIDX']; data_off,_ = chunks['DATA']
    media = []
    for i in range(0,size,12):
        mid,start,length = struct.unpack_from('<III',blob,off+i)
        media.append((mid,blob[data_off+start:data_off+start+length]))
    out = ROOT/'Platforms/Windows/Audio';out.mkdir(parents=True,exist_ok=True)
    (out/(NAME+'.bnk')).write_bytes(b.assemble(NAME,events,media))
    (out/'MilitaryParadeBanks.ini').write_bytes(('\r\n'.join([
        '[Global]','','[Menu]','','[InGame]','!'+NAME+'.bnk','','[2D]','','[3D]','','[FMV]','!'+NAME+'.bnk',''])).encode())
    print('Built native v113 bank:',PLAY,STOP)

if __name__ == '__main__': build()

import hashlib,shutil,sys,xml.etree.ElementTree as E

def build(destination):
 source=Path(__file__).resolve().parents[1]
 destination=Path(destination).resolve()
 if destination.is_relative_to(source):raise ValueError('Output must be outside the source tree')
 if destination.exists():raise FileExistsError('Choose a new output directory; existing files will not be overwritten')
 destination.mkdir(parents=True)
 release=destination/'Release/MilitaryParade';release.mkdir(parents=True)
 manifest=E.parse(source/'MilitaryParade.modinfo')
 names=['MilitaryParade.modinfo']+[n.text for n in manifest.findall('./Files/File')]
 assert len(names)==len(set(names))
 for name in names:
  rel=Path(name)
  if rel.is_absolute()or'..'in rel.parts:raise ValueError('Unsafe manifest path')

 actual={p.relative_to(release).as_posix()for p in release.rglob('*')if p.is_file()}

 shutil.copytree(source,destination/'Source/MilitaryParade',ignore=shutil.ignore_patterns('__pycache__','*.pyc','.git'))
 docs=destination/'Docs';docs.mkdir()
 for name in ['README.md','CHANGELOG.md','VALIDATION.md','WORKSHOP_DESCRIPTION_zh_CN.md']:

"""Inspect a built/installed mod folder; never modifies it.
Usage: python tools/verify_build.py /path/to/the/built/mod
This checks files, not whether the game loaded them.
"""
from pathlib import Path
import sys,xml.etree.ElementTree as E

def verify(folder):
 root=Path(folder).resolve()
 mod=E.parse(root/'MilitaryParade.modinfo')
 assert mod.getroot().get('version')=='22','Build is not 3.20 B1 (mod version 22)'
 action=mod.find('./InGameActions/UpdateArt')
 assert action is not None,'UpdateArt missing from built manifest'
 dep_name=action.findtext('File')
 assert dep_name and dep_name!='(Mod Art Dependency File)','SDK did not resolve the art dependency placeholder'
 dep=(root/dep_name).read_text(encoding='utf-8-sig')
 assert 'Camera.artdef'in dep and 'Camera'in dep,'built dependency does not register Camera.artdef'
 art=(root/'ArtDefs/Camera.artdef').read_text(encoding='utf-8-sig')
 assert 'AssetObjects::ArtDefSet'in art,'Windows SDK ArtDef type spelling missing'
 camera=E.fromstring(art.replace('AssetObjects::','AssetObjects..'))
 heights=[]
 for item in camera.findall('m_RootCollections/Element/Element'):
  if item.find('m_Name').get('text')!='DEFAULT_CAMERA':continue
  for coll in item.findall('m_ChildCollections/Element'):
   if coll.find('m_CollectionName').get('text')!='HeightCurve':continue
   for point in coll.findall('Element'):
    values={v.find('m_ParamName').get('text'):float(v.findtext('m_fValue'))for v in point.findall('m_Fields/m_Values/Element')}
    if values['Time']==0:heights.append(values['Height'])
 assert heights==[65.0],f'built near camera height is {heights}, expected 65'
 replacement=next((a.find('Properties/LuaReplace').text for a in mod.findall('./InGameActions/ReplaceUIScript')if a.findtext('Properties/LuaContext')=='HistoricMoments'),None)
 assert replacement,'HistoricMoments replacement missing from built manifest'
 assert "diagnostic.TimelineBuild='3.20 B1'"in(root/replacement).read_text(encoding='utf-8-sig'),'old historic-moment Lua in output'
 assert 'PrideMoments_ToggleTimeline(false)'in(root/'Parade_UI.lua').read_text(encoding='utf-8-sig'),'old UI dispatch in output'
 for file in mod.findall('./Files/File'):assert(root/file.text).is_file(),file.text
 print('PASS: 3.20 B1 manifest, resolved art dependency, height 65, timeline replacement, native entry, all listed runtime files')

if __name__=='__main__':
 if len(sys.argv)!=2:raise SystemExit(__doc__)
 verify(sys.argv[1])
